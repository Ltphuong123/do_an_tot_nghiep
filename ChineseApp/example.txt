// 2. Component Biểu đồ Tròn (Donut Chart) - Fix Domino Logic
const DonutChart = ({ data, total }) => {
  const radius = 90; 
  const stroke = 22; 
  const normalizedRadius = radius - stroke / 2;
  const circumference = normalizedRadius * 2 * Math.PI;
  
  // Tính toán segments
  let currentOffset = 0;
  // Thứ tự hiển thị: Xanh -> Đỏ -> Vàng -> Xanh dương
  const rawSegments = [
    { color: '#10b981', value: data.memorized }, // Green
    { color: '#ef4444', value: data.notMemorized }, // Red
    { color: '#f59e0b', value: data.unsure }, // Yellow
    { color: '#3b82f6', value: data.favorite }, // Blue
  ];

  // Lọc bỏ các phần tử = 0 để tránh lỗi hiển thị và tính toán
  const activeSegments = rawSegments.filter(s => s.value > 0);

  const calculatedSegments = activeSegments.map((segment) => {
    const strokeDashoffset = circumference - (segment.value / total) * circumference;
    const rotation = currentOffset;
    currentOffset += (segment.value / total) * 360;
    
    return {
      ...segment,
      strokeDashoffset,
      rotation
    };
  });

  return (
    <div className="relative flex items-center justify-center w-[180px] h-[180px]">
      <svg height={radius * 2} width={radius * 2} className="rotate-[-90deg]">
        {/* Track nền */}
        <circle
          stroke="currentColor"
          strokeWidth={stroke}
          fill="transparent"
          r={normalizedRadius}
          cx={radius}
          cy={radius}
          className="text-gray-100 dark:text-zinc-800 transition-colors duration-500" 
        />
        
        {/* Vẽ các segments chính (Màu sau đè lên đuôi màu trước) */}
        {calculatedSegments.map((segment, index) => (
          <circle
            key={index}
            stroke={segment.color}
            strokeWidth={stroke}
            strokeDasharray={`${circumference} ${circumference}`}
            style={{ 
              strokeDashoffset: segment.strokeDashoffset,
              transformOrigin: 'center',
              transform: `rotate(${segment.rotation}deg)`
            }}
            strokeLinecap="round" 
            fill="transparent"
            r={normalizedRadius}
            cx={radius}
            cy={radius}
            className="transition-all duration-1000 ease-out"
          />
        ))}

        {/* DOMINO FIX: Vẽ lại "đầu" của segment ĐẦU TIÊN lên lớp trên cùng.
            Điều này làm cho đầu của màu đầu tiên đè lên đuôi của màu cuối cùng,
            hoàn thiện vòng lặp vô cực. */}
        {calculatedSegments.length > 1 && (
            <circle
                stroke={calculatedSegments[0].color}
                strokeWidth={stroke}
                // Chỉ vẽ 1 chấm rất nhỏ (0.1) để tạo nắp tròn
                strokeDasharray={`0.1 ${circumference}`} 
                style={{ 
                    transformOrigin: 'center',
                    transform: `rotate(${calculatedSegments[0].rotation}deg)` 
                }}
                strokeLinecap="round" 
                fill="transparent"
                r={normalizedRadius}
                cx={radius}
                cy={radius}
            />
        )}
      </svg>
      
      {/* Text ở giữa */}
      <div className="absolute flex flex-col items-center justify-center pt-1 text-gray-800 dark:text-white">
        <span className="text-4xl font-semibold tracking-tight leading-none">{total}</span>
        <span className="text-base font-medium opacity-80 mt-1">từ</span>
      </div>
    </div>
  );
};

// 3. Component Biểu đồ Cột (Bar Chart) - Full Width (Giữ nguyên)
const BarChartFull = ({ data }) => {
  const maxVal = Math.max(data.memorized, data.notMemorized, data.unsure, data.favorite, 10); 

  const bars = [
    { label: 'Thuộc', value: data.memorized, color: 'bg-emerald-500', icon: CheckCircle2, text: 'text-emerald-500' },
    { label: 'Chưa', value: data.notMemorized, color: 'bg-red-500', icon: XCircle, text: 'text-red-500' },
    { label: 'K.Chắc', value: data.unsure, color: 'bg-amber-500', icon: HelpCircle, text: 'text-amber-500' },
    { label: 'Thích', value: data.favorite, color: 'bg-blue-500', icon: Heart, text: 'text-blue-500' },
  ];

  return (
    <div className="w-full h-full flex flex-col justify-end">
      <div className="flex items-end justify-between gap-3 h-[136px] px-2"> 
        {bars.map((bar, idx) => (
            <div key={idx} className="relative flex flex-col items-center justify-end h-full flex-1 group">
                <div 
                    className={`w-full rounded-t-md ${bar.color} opacity-90 transition-all duration-500 relative`}
                    style={{ height: `${(bar.value / maxVal) * 100}%`, minHeight: '8px' }}
                >
                </div>
            </div>
        ))}
      </div>

      <div className="flex justify-between gap-2 mt-3 px-1 border-t border-gray-200 dark:border-zinc-800 pt-2">
         {bars.map((bar, idx) => (
             <div key={idx} className="flex flex-col items-center flex-1">
                 <span className={`text-base font-bold ${bar.text}`}>{bar.value}</span>
                 <div className="flex items-center gap-1 opacity-80 mt-0.5">
                     <span className="text-[10px] uppercase font-bold tracking-wider truncate">{bar.label}</span>
                 </div>
             </div>
         ))}
      </div>
    </div>
  );
};
